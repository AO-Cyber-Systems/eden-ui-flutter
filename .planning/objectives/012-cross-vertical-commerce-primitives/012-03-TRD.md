<objective>
Ship `EdenPaymentEntry` — payment-method picker + amount + reference + note entry. Configurable allowed-methods allowlist (consumer constrains POS to cash+card+gift, medical to card+check+portal, trades to card+ACH+check, fuel POD to cash+card+account). Returns `EdenPaymentDraft` value object via `onDraftChanged` callback. Consumer wires actual payment processing.

Per validation §4.1 row 3: appears in M5 · F3 · R1 · T2 · T5. Five blocked screens.

Depends on 012-01 because 012-01 bootstraps `commerce_screen.dart`; this TRD APPENDS a section. Independent of 012-02.

**Transport-agnostic.** This widget does NOT process payments. It produces a draft value object. Downstream consumer code (in `eden-biz-flutter` or vertical apps) wires the actual payment processor.
</objective>

<anti_patterns>
- **Do NOT include a 'Submit' button.** Real-time `onDraftChanged` callback emits every change. Consumer renders submit UI separately.
- **Do NOT validate that amount > 0.** Some flows allow $0.00 payments (gift-card-only checkout, full discount).
- **Do NOT format the amount in the TextField as currency on every keystroke.** Cursor jumping is unfixable when the formatter rewrites text mid-edit. Show raw decimal in the field; the consumer (or 012-04 EdenSplitTender summary row) formats display values.
- **Do NOT eat method chip taps when the keyboard is open.** Use `FocusScope.of(context).unfocus()` BEFORE setting the new method to dismiss keyboard cleanly.
- **Do NOT bind to specific payment processor SDKs.** Library exposes intent (draft). Stripe, Square, Adyen integration is downstream concern.
</anti_patterns>

class EdenPaymentDraft {
  const EdenPaymentDraft({
    required this.method,
    required this.amount,
    this.reference,
    this.note,
    this.capturedAt,
  });
  final EdenPaymentMethod method;
  final double amount;
  final String? reference;
  final String? note;
  final DateTime? capturedAt;

  EdenPaymentDraft copyWith({
    EdenPaymentMethod? method, double? amount, String? reference, String? note, DateTime? capturedAt,
  }) => EdenPaymentDraft(
    method: method ?? this.method,
    amount: amount ?? this.amount,
    reference: reference ?? this.reference,
    note: note ?? this.note,
    capturedAt: capturedAt ?? this.capturedAt,
  );

class EdenPaymentEntry extends StatefulWidget {
  const EdenPaymentEntry({
    super.key,
    required this.allowedMethods,
    required this.onDraftChanged,
    this.initialDraft,
    this.expectedAmount,
    this.currencyCode = 'USD',
    this.requireReference = false,
  });
  final List<EdenPaymentMethod> allowedMethods;
  final ValueChanged<EdenPaymentDraft> onDraftChanged;
  final EdenPaymentDraft? initialDraft;
  final double? expectedAmount;
  final String currencyCode;
  final bool requireReference;

  @override
  State<EdenPaymentEntry> createState() => _EdenPaymentEntryState();
}

class _EdenPaymentEntryState extends State<EdenPaymentEntry> {
  late EdenPaymentMethod _method;
  late TextEditingController _amountController;
  late TextEditingController _referenceController;
  late TextEditingController _noteController;

  @override
  void initState() {
    super.initState();
    _method = widget.initialDraft?.method ?? widget.allowedMethods.first;
    _amountController = TextEditingController(
      text: widget.initialDraft?.amount.toString() ?? '');
    _referenceController = TextEditingController(
      text: widget.initialDraft?.reference ?? '');
    _noteController = TextEditingController(
      text: widget.initialDraft?.note ?? '');
  }

  @override
  void dispose() {
    _amountController.dispose();
    _referenceController.dispose();
    _noteController.dispose();
    super.dispose();
  }

  String _currencySymbol() {
    try {
      return intl.NumberFormat.simpleCurrency(name: widget.currencyCode).currencySymbol;
    } catch (_) {
      return widget.currencyCode;
    }
  }

  bool _showReference() {
    if (_method == EdenPaymentMethod.cash) return false;
    if (_method == EdenPaymentMethod.accountOnFile || _method == EdenPaymentMethod.other) {
      return widget.requireReference;
    }
    return true;
  }

  String _referenceHint() => switch (_method) {
    EdenPaymentMethod.card => 'Last 4 of card',
    EdenPaymentMethod.ach => 'Account last 4',
    EdenPaymentMethod.check => 'Check #',
    EdenPaymentMethod.portal => 'Confirmation #',
    EdenPaymentMethod.giftCard => 'Gift card #',
    EdenPaymentMethod.accountOnFile => 'Account ID',
    EdenPaymentMethod.other => 'Reference',
    _ => '',
  };

  void _emit() {
    final amount = double.tryParse(_amountController.text) ?? 0.0;
    final reference = _referenceController.text.isEmpty ? null : _referenceController.text;
    final note = _noteController.text.isEmpty ? null : _noteController.text;
    widget.onDraftChanged(EdenPaymentDraft(
      method: _method, amount: amount, reference: reference, note: note,
    ));
  }

  @override
  Widget build(BuildContext context) {
    // ... Wrap-of-chips + amount field + reference field + note field + optional mismatch banner.

  }
}

Widget _amountMismatchBanner() {
  final amount = double.tryParse(_amountController.text) ?? 0.0;
  final diff = (amount - widget.expectedAmount!).abs();
  if (diff <= 0.01) return const SizedBox.shrink();
  final expected = intl.NumberFormat.currency(name: widget.currencyCode).format(widget.expectedAmount!);
  return Padding(
    padding: const EdgeInsets.only(top: EdenSpacing.space2),
    child: EdenBanner(
      message: 'Amount differs from expected: $expected',
      severity: EdenBannerSeverity.warning,
    ),

}
