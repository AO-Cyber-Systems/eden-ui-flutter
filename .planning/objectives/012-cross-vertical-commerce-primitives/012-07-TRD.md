```dart
class EdenDonutChart extends StatelessWidget {
  const EdenDonutChart({
    super.key,
    required this.data,
    this.size = 200,
    this.donutWidth = 40,
    this.centerLabel,
    this.centerLabelSlot,
    this.showLegend = true,
    this.legendPosition = EdenChartLegendPosition.bottom,
    this.colors,
  });
  // ... fields
  @override
  Widget build(BuildContext context) {
    final chart = EdenPieChart(
      data: data, size: size, donut: true, donutWidth: donutWidth,
      centerLabel: centerLabelSlot == null ? centerLabel : null,
      centerLabelSlot: centerLabelSlot,
      showLegend: false, // we control legend rendering for positioning
      showLabels: false,
      colors: colors,
    );

    if (!showLegend) return chart;

    final legend = _buildLegend(context);
    return switch (legendPosition) {
      EdenChartLegendPosition.bottom => Column(children: [chart, const SizedBox(height: 8), legend]),
      EdenChartLegendPosition.right => Row(children: [chart, const SizedBox(width: 16), Expanded(child: legend)]),
    };
  }

  Widget _buildLegend(BuildContext context) {
    final palette = colors ?? _kDefaultPalette;
    return Wrap(
      spacing: 12, runSpacing: 4,
      children: [
        for (int i = 0; i < data.length; i++)
          Row(mainAxisSize: MainAxisSize.min, children: [
            Container(width: 12, height: 12, color: palette[i % palette.length]),
            const SizedBox(width: 4),
            Text(data[i].label),
          ]),
      ],
    );
  }
}
```

<anti_patterns>
- **Do NOT rewrite EdenPieChart.** EdenDonutChart is a composition shorthand. The work is: (a) add `centerLabelSlot: Widget?` additive param to EdenPieChart, (b) create EdenDonutChart wrapping EdenPieChart, (c) add legend positioning at the EdenDonutChart level (above EdenPieChart's internal legend).
- **Do NOT remove `donut: bool` from EdenPieChart.** Existing call sites with `donut: true` continue to work.
- **Do NOT add tooltips.** Out of scope.
- **Do NOT centralize the legend inside EdenPieChart and have EdenDonutChart hide it.** Cleaner approach: pass `showLegend: false` to the inner EdenPieChart, render OUR legend at the EdenDonutChart level (where we have full control over positioning). Keep EdenPieChart's legend code intact for non-donut consumers.
</anti_patterns>

<gotchas>
- **`centerLabelSlot` in EdenPieChart implementation.** When non-null, wrap the existing `CustomPaint` in a `Stack` and `Center` the widget. Conditionally render based on which is provided (centerLabel String vs centerLabelSlot Widget). Both null = no center content.
- **Legend width sizing.** When `legendPosition: right`, the legend's `Wrap` needs a bounded width or it'll try to fill infinite space. Wrap in `Expanded` inside the Row (already in pseudocode above).
- **Empty data legend.** If `data: []`, legend renders empty Wrap. No-op visually. No exception.
- **Default palette access.** `_kDefaultPalette` is the const list at the top of eden_chart.dart (lines ~33-40). It's file-private (no underscore-trick needed — same file, full access from EdenDonutChart).
- **commerce_screen.dart APPEND ONLY.** Replace `// TRD 012-07 appends: ...` placeholder.
</gotchas>

<error_recovery>
- **If existing EdenPieChart's internal legend is tangled with the pie painter** and can't be cleanly suppressed via `showLegend: false`: pass through anyway; the legend may double-render but our higher-level legend takes visual precedence. Document as known minor visual issue. Or as a safer move: temporarily wrap EdenPieChart in a `MediaQuery(data: ...)` that constrains the height to just the donut canvas, hiding any internal legend.
- **If centerLabelSlot causes layout issues** because Stack siblings can't share `size`: use `Positioned.fill(child: Center(child: centerLabelSlot))` inside the Stack.
- **If legendPosition.right at iPhone-narrow makes the chart too small:** acceptable. Catalog scenario for legendPosition.right uses width > 600pt. Document that `.right` is recommended only for wide layouts.
</error_recovery>

```dart
class EdenDonutChart extends StatelessWidget {
  const EdenDonutChart({
    super.key,
    required this.data,
    this.size = 200,
    this.donutWidth = 40,
    this.centerLabel,
    this.centerLabelSlot,
    this.showLegend = true,
    this.legendPosition = EdenChartLegendPosition.bottom,
    this.colors,
  });
  final List<EdenChartDataPoint> data;
  final double size;
  final double donutWidth;
  final String? centerLabel;
  final Widget? centerLabelSlot;
  final bool showLegend;
  final EdenChartLegendPosition legendPosition;
  final List<Color>? colors;

  @override
  Widget build(BuildContext context) {
    final chart = EdenPieChart(
      data: data,
      size: size,
      donut: true,
      donutWidth: donutWidth,
      centerLabel: centerLabelSlot == null ? centerLabel : null,
      centerLabelSlot: centerLabelSlot,
      showLabels: false,
      showLegend: false, // we render our own legend for positioning control
      colors: colors,
    );

    if (!showLegend || data.isEmpty) return chart;

    final palette = colors ?? _kDefaultPalette;
    final legend = Wrap(
      spacing: 12,
      runSpacing: 4,
      children: [
        for (int i = 0; i < data.length; i++)

            Container(width: 12, height: 12, color: palette[i % palette.length]),
            const SizedBox(width: 4),
            Text(data[i].label),
          ]),
      ],
    );

    return switch (legendPosition) {
      EdenChartLegendPosition.bottom => Column(
          mainAxisSize: MainAxisSize.min,
          children: [chart, const SizedBox(height: 8), legend],
        ),
      EdenChartLegendPosition.right => Row(
          mainAxisSize: MainAxisSize.min,
          children: [chart, const SizedBox(width: 16), Flexible(child: legend)],
```
