<gotchas>
- **NaN handling in painter.** When `nullablePoints: true`, painter must skip NaN entries — for each segment, check `if (a.isNaN || b.isNaN) continue;`. Otherwise the line connects past NaN as if it were 0.
- **referenceLines y-position computation.** With auto min/max, a reference line at y=12.0 maps to canvas y = `size.height - (12.0 - minY) / (maxY - minY) * size.height`. With explicit minValue/maxValue, use those instead.
- **Dashed line rendering.** Flutter Canvas has no built-in dash API. Use `Path` + a small helper that walks segments adding 4pt dash + 3pt gap. Or skip dashed and use solid 1pt with 30% opacity — simpler, equally effective visually.
- **width=double.infinity inside Row.** `CustomPaint(size: Size(double.infinity, height))` inside Row → Row needs Expanded ancestor for the sparkline to bound. Test cases must wrap in Expanded or fixed-width SizedBox.

```dart
    super.key,
    required this.values,
    this.height = 40,
    this.width,
    this.color,
    this.showArea = true,
    this.lineWidth = 2,
    this.minValue,
    this.maxValue,
```

```dart
  });

  final List<double> values;
  final double height;
  final double? width;
  final Color? color;
  final bool showArea;
  final double lineWidth;
  final double? minValue;
  final double? maxValue;
  final List<double> referenceLines;
  final bool nullablePoints;

  @override

    final theme = Theme.of(context);
    final c = color ?? theme.colorScheme.primary;
    return CustomPaint(
      size: Size(width ?? double.infinity, height),
      painter: _SparklinePainter(
        values: values,
        color: c,
        showArea: showArea,

        minValueOverride: minValue,

        nullablePoints: nullablePoints,
      ),
    );
  }
}

class _SparklinePainter extends CustomPainter {
  _SparklinePainter({
    required this.values,
    required this.color,
    required this.showArea,
    required this.lineWidth,
    this.minValueOverride,
    this.maxValueOverride,
    this.referenceLines = const [],
    this.nullablePoints = false,
  });

  final List<double> values;
  final Color color;
  final bool showArea;
  final double lineWidth;
  final double? minValueOverride;
  final double? maxValueOverride;
  final List<double> referenceLines;
  final bool nullablePoints;

  double _minY() {
    if (minValueOverride != null) return minValueOverride!;
    final filtered = values.where((v) => !v.isNaN);
    if (filtered.isEmpty) return 0;
    return filtered.reduce(math.min);
  }

  double _maxY() {
    if (maxValueOverride != null) return maxValueOverride!;
    final filtered = values.where((v) => !v.isNaN);
    if (filtered.isEmpty) return 1;
    return filtered.reduce(math.max);
  }

  @override
  void paint(Canvas canvas, Size size) {
    if (values.isEmpty) return;
    final minY = _minY();
    final maxY = _maxY();
    final range = (maxY - minY).abs() < 1e-9 ? 1.0 : (maxY - minY);

    // 1. Reference lines first (background, behind data line).
    if (referenceLines.isNotEmpty) {
      final refPaint = Paint()
        ..color = color.withValues(alpha: 0.30)
        ..strokeWidth = 1
        ..style = PaintingStyle.stroke;
      for (final ref in referenceLines) {
        final y = size.height - (ref - minY) / range * size.height;
        canvas.drawLine(Offset(0, y), Offset(size.width, y), refPaint);
      }
    }

    // 2. Data line (preserve existing area-fill logic; add NaN skipping when nullablePoints true).
    // ... existing painter logic adapted to skip NaN segments when nullablePoints true.
  }

  @override
  bool shouldRepaint(_SparklinePainter old) =>
      old.values != values ||
      old.color != color ||
      old.showArea != showArea ||
      old.lineWidth != lineWidth ||
      old.minValueOverride != minValueOverride ||
      old.maxValueOverride != maxValueOverride ||
      old.referenceLines != referenceLines ||
      old.nullablePoints != nullablePoints;
}
```
