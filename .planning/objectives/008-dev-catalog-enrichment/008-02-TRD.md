class _TradesJobsListDemo extends StatelessWidget {
  const _TradesJobsListDemo();
  @override
  Widget build(BuildContext context) {
    final jobs = TradesScenarios.weekEvents;
    return SizedBox(
      height: 480,
      child: EdenListPageScaffold(
        title: 'Trades jobs (this week)',
        createLabel: 'New job',
        onCreate: () {},
        filterPills: Wrap(spacing: 6, children: const [
          EdenBadge(label: 'All', variant: EdenBadgeVariant.primary),
          EdenBadge(label: 'Open', variant: EdenBadgeVariant.secondary),
          EdenBadge(label: 'Scheduled'),
          EdenBadge(label: 'Completed'),
        ]),
        searchHint: 'Search jobs, customers, addresses...',
        onSearchChanged: (_) {},
        alerts: EdenBlockingAlerts(alerts: TradesScenarios.blockingAlerts),
        body: ListView.builder(
          itemCount: jobs.length,
          itemBuilder: (_, i) => _TradesJobRow(event: jobs[i]),
        ),
      ),
    );
  }
}

class _TradesJobRow extends StatelessWidget {
  const _TradesJobRow({required this.event});
  final EdenSchedulerEvent event;

  @override
  Widget build(BuildContext context) {
    return ListTile(
      leading: const Icon(Icons.build),
      title: Text(event.title),
      subtitle: Text('${event.assignee ?? ""} · ${event.status ?? ""}'),
      trailing: const EdenUrgencyBadge(urgency: 'high'),
    );
  }
}
```

  Widget build(BuildContext context) {

    return SizedBox(
      height: 480,
      child: EdenListPageScaffold(

          itemBuilder: (_, i) => _TradesJobRow(event: jobs[i]),

      ),
    );
  }
}

class _ResponsiveListScaffoldDemo extends StatelessWidget {
  const _ResponsiveListScaffoldDemo();
  @override
  Widget build(BuildContext context) {
    return SingleChildScrollView(
      scrollDirection: Axis.horizontal,
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          _ResponsiveColumn(width: 390, label: 'Compact (390pt)'),
          const SizedBox(width: 24),
          _ResponsiveColumn(width: 768, label: 'Medium (768pt)'),
          const SizedBox(width: 24),
          _ResponsiveColumn(width: 1280, label: 'Expanded (1280pt)'),
        ],
      ),
    );
  }
}

class _ResponsiveColumn extends StatelessWidget {
  const _ResponsiveColumn({required this.width, required this.label});
  final double width; final String label;
  @override
  Widget build(BuildContext context) {
    return Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
      Text(label, style: Theme.of(context).textTheme.labelMedium),
      const SizedBox(height: 4),
      SizedBox(
        width: width,
        height: 360,
        child: const _TradesJobsListDemo(),
      ),
    ]);
  }
}

// test/dev_app/layouts_screen_test.dart
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:eden_ui_flutter/dev_app/screens/layouts_screen.dart';

void main() {
  testWidgets('LayoutsScreen renders at iPhone-narrow (390pt) without overflow',
      (tester) async {
    tester.view.physicalSize = const Size(390, 844);
    tester.view.devicePixelRatio = 1.0;
    addTearDown(tester.view.resetPhysicalSize);
    addTearDown(tester.view.resetDevicePixelRatio);

    await tester.pumpWidget(
      const MaterialApp(home: LayoutsScreen()),
    );
    await tester.pumpAndSettle();

    // Assert no exception was thrown during layout.
    expect(tester.takeException(), isNull);
    // Smoke-assert: at least the app bar title rendered.
    expect(find.text('Layouts'), findsOneWidget);
  });

  testWidgets('LayoutsScreen scaffolds preview shows cross-vertical demos',
      (tester) async {
    tester.view.physicalSize = const Size(1280, 800);
    addTearDown(tester.view.resetPhysicalSize);

    await tester.pumpWidget(
      const MaterialApp(home: LayoutsScreen()),
    );
    await tester.pumpAndSettle();

    // Click into the Scaffolds preview tab if there's a toggle.
    // (If LayoutsScreen always starts on scaffolds, skip.)
    // Assert at least one of the vertical demo section titles renders.
    expect(find.textContaining('Trades jobs'), findsAtLeastNWidgets(1));
  });
}
```
