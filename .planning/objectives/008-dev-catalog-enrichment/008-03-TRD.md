```dart
class _TradesJobsTableDemo extends StatelessWidget {
  const _TradesJobsTableDemo();
  @override
  Widget build(BuildContext context) {
    final jobs = TradesScenarios.weekEvents;
    return SingleChildScrollView(
      scrollDirection: Axis.horizontal,
      child: EdenDataTable(
        columns: const [
          EdenTableColumn(label: 'Job ID'),
          EdenTableColumn(label: 'Customer', flex: 2),
          EdenTableColumn(label: 'Type'),
          EdenTableColumn(label: 'Truck'),
          EdenTableColumn(label: 'Status'),
          EdenTableColumn(label: 'Urgency'),
          EdenTableColumn(label: 'Start'),
          EdenTableColumn(label: 'Duration'),
        ],
        rows: [
          for (final e in jobs)
            EdenTableRow(cells: [
              Text(e.id, style: const TextStyle(fontFamily: 'monospace')),
              Text(e.title),
              Text(e.type ?? '—'),
              Text(e.assignee ?? '—'),
              EdenBadge(label: e.status ?? '—', variant: EdenBadgeVariant.secondary, size: EdenBadgeSize.sm),
              const EdenUrgencyBadge(urgency: 'medium'),
              Text('${e.start.month}/${e.start.day} ${e.start.hour}:00'),
              Text('${e.end.difference(e.start).inMinutes}m'),
            ]),
        ],
      ),
    );
  }
}

class _MixedActivityFeedDemo extends StatelessWidget {
  const _MixedActivityFeedDemo();
  @override
  Widget build(BuildContext context) {
    final feed = CrossCuttingFixtures.mixedActivityFeed;
    return EdenCard(
      child: Column(children: [
        for (final e in feed) EdenActivityFeedItem(data: e),
      ]),
    );
  }
}

class _FuelTankLevelsDemo extends StatelessWidget {
  const _FuelTankLevelsDemo();
  @override
  Widget build(BuildContext context) {
    final tanks = FuelScenarios.tankLevels;
    return Column(children: [
      for (final t in tanks)
        Padding(
          padding: const EdgeInsets.only(bottom: EdenSpacing.space2),
          child: Row(children: [
            Expanded(flex: 2, child: Text(t.terminalName)),
            Expanded(flex: 3, child: EdenStockLevelIndicator(
              percentage: t.percentage,
              isBelowReorder: t.isBelowReorder,
            )),
            const SizedBox(width: 8),
            Text('${(t.percentage * 100).round()}%',
                 style: const TextStyle(fontFamily: 'monospace')),
          ]),
        ),
    ]);
  }
}
```

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:eden_ui_flutter/dev_app/screens/data_display_screen.dart';

void main() {
  testWidgets('DataDisplayScreen renders at iPhone-narrow without overflow',
      (tester) async {
    tester.view.physicalSize = const Size(390, 844);
    tester.view.devicePixelRatio = 1.0;
    addTearDown(tester.view.resetPhysicalSize);
    addTearDown(tester.view.resetDevicePixelRatio);

    await tester.pumpWidget(
      const MaterialApp(home: DataDisplayScreen()),
    );
    await tester.pumpAndSettle();

    expect(tester.takeException(), isNull);
    expect(find.text('Data Display'), findsOneWidget);
    // Confirm at least one cross-vertical demo title rendered.
    expect(find.textContaining('Trades'), findsAtLeastNWidgets(1));
  });
}
```
