/// A parity comparison row: trades-react reference image on one side,
/// live EdenScheduler in matching view mode on the other.
class _ParityRow extends StatelessWidget {
  const _ParityRow({
    required this.label,
    required this.referenceAsset,
    required this.view,
    required this.controller,
    required this.events,
    required this.assignees,
    this.forceMobile = false,
  });

  final String label;
  final String referenceAsset;
  final EdenSchedulerView view;
  final EdenSchedulerController controller;
  final List<EdenSchedulerEvent> events;
  final List<String> assignees;
  final bool forceMobile;

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(builder: (context, c) {
      final isCompact = c.maxWidth < 900;
      // Each side gets its own controller copy so the comparison view doesn't
      // move when the user changes the top-level controller.
      final localController = EdenSchedulerController(
        initialView: view,
        initialDate: DateTime(2026, 5, 16),
      );
      Widget liveScheduler = SizedBox(
        height: 480,
        child: AnimatedBuilder(
          animation: localController,
          builder: (context, _) {
            return EdenScheduler(
              controller: localController,
              events: events,
              assignees: assignees,
              selectedAssignees: const {},
              forceMobileView: forceMobile,
            );
          },
        ),
      );
      Widget refImage = Container(
        height: 480,
        decoration: BoxDecoration(border: Border.all(color: Colors.grey)),
        child: Image.asset(referenceAsset, fit: BoxFit.contain),
      );

      Widget body;
      if (isCompact) {
        body = Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
          const Text('trades-react reference', style: TextStyle(fontWeight: FontWeight.w600)),
          const SizedBox(height: 4),
          refImage,
          const SizedBox(height: 12),
          const Text('Live EdenScheduler', style: TextStyle(fontWeight: FontWeight.w600)),
          const SizedBox(height: 4),
          liveScheduler,
        ]);
      } else {
        body = Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
          Expanded(child: Column(children: [
            const Text('trades-react reference', style: TextStyle(fontWeight: FontWeight.w600)),
            const SizedBox(height: 4),
            refImage,
          ])),
          const SizedBox(width: 12),
          Expanded(child: Column(children: [
            const Text('Live EdenScheduler', style: TextStyle(fontWeight: FontWeight.w600)),
            const SizedBox(height: 4),
            liveScheduler,
          ])),
        ]);
      }
      return Padding(
        padding: const EdgeInsets.only(bottom: EdenSpacing.space6),
        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
          Text(label, style: Theme.of(context).textTheme.titleMedium),
          const SizedBox(height: 8),
          body,
        ]),
      );
    });
  }
}
