class _UrgencyContextDemo extends StatelessWidget {
  const _UrgencyContextDemo();
  @override
  Widget build(BuildContext context) {
    const rows = [
      _UrgencyCtx('Trades', 'AC out · 1 child + 1 elderly resident', 'critical'),
      _UrgencyCtx('Salon', 'Bridal updo · ceremony in 3h', 'high'),
      _UrgencyCtx('Medical', 'Walk-in · chest pain · 67yo male', 'critical'),
      _UrgencyCtx('Gov', 'SLA breach · case 30d overdue', 'high'),
      _UrgencyCtx('Fuel', 'Tank at 8% · industrial customer', 'critical'),
      _UrgencyCtx('Trades', 'Annual maintenance · scheduled', 'low'),
      _UrgencyCtx('Salon', 'Bang trim · drop-in', 'low'),
    ];
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [for (final r in rows) _UrgencyContextRow(ctx: r)],
    );
  }
}

class _UrgencyCtx {
  final String vertical, label, urgency;
  const _UrgencyCtx(this.vertical, this.label, this.urgency);
}

class _UrgencyContextRow extends StatelessWidget {
  const _UrgencyContextRow({required this.ctx});
  final _UrgencyCtx ctx;
  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 8),
      child: Row(children: [
        SizedBox(width: 60, child: Text(ctx.vertical,
            style: const TextStyle(fontFamily: 'monospace', fontSize: 11))),
        const SizedBox(width: 8),
        Expanded(child: Text(ctx.label)),
        const SizedBox(width: 8),
        EdenUrgencyBadge(urgency: ctx.urgency),
      ]),
    );
  }
}

class _CustomerTierListDemo extends StatelessWidget {
  const _CustomerTierListDemo();
  @override
  Widget build(BuildContext context) {
    return Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
      for (final c in CrossCuttingFixtures.customers) Padding(
        padding: const EdgeInsets.only(bottom: 6),
        child: Row(children: [
          Expanded(child: Text(c.name)),
          c.tier == null
            ? const SizedBox.shrink()
            : EdenMembershipTierBadge(tier: c.tier!),
        ]),
      ),
    ]);
  }
}
