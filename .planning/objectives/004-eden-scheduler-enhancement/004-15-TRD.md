Wave 4 — Dialogs. Generic slot-based dialogs for create / edit / detail. The library doesn't know your domain (appointment form fields, customer pickers, etc.) — it provides the dialog scaffold (responsive sizing, cancel-button + primary-action-button, top app bar with title, scrollable body) and a `bodySlot: Widget` for the consumer's domain content.

<context>
**Locked design inputs:**
- Donor `AppointmentEditDialog.tsx`, `AppointmentDetailDialog.tsx`, `AppointmentCompletionDialog.tsx`, `CreateAppointmentDialog.tsx` — each is a Material Dialog wrapper around the consumer's form.
- Library version is generic: dialog shape + cancel/save buttons + bodySlot for consumer form.
- Out of scope for library: the actual form fields (customer picker, location picker, truck picker, type picker, etc.) — those are biz-coupled and belong in eden-biz-flutter.

**Approach:**
- 3 widgets sharing a common scaffold:
  - `EdenSchedulerCreateDialog` — for new events.
  - `EdenSchedulerEditDialog` — for existing events (accepts `initialEvent`).
  - `EdenSchedulerDetailDialog` — read-only view (no save button, just close + optional action slot like "Edit" or "Delete").

**Responsive sizing:** at >= 600pt width, render as a modal Dialog (max 480pt wide). At < 600pt, fullscreen Scaffold (Dialog.fullscreen).

**Static `.show()` method:**
```dart
static Future<T?> show<T>({
  required BuildContext context,
  required String title,
  required Widget bodySlot,
  required String primaryActionLabel,
  required VoidCallback? onPrimaryAction,
  String cancelLabel = 'Cancel',
});
```

Returns the `Navigator.pop` result, so consumer can `final result = await EdenSchedulerCreateDialog.show(...)` and act on saved-form-data.
</context>

<gotchas>
- **Form state is consumer's responsibility.** Library doesn't manage form. Consumer creates a TextEditingController + FormKey, builds a Form widget passed as bodySlot, validates + pops with data on Save.
- **Dialog scrollability:** the bodySlot may be tall — wrap in `SingleChildScrollView` automatically. Consumer doesn't need to handle.
- **Mobile fullscreen vs desktop modal:** `Dialog.fullscreen(child: Scaffold(appBar: AppBar(title: Text(title), actions: [TextButton('Save')], leading: CloseButton()), body: bodySlot))` for mobile. `Dialog(child: ConstrainedBox(maxWidth: 480, child: Column(...)))` for desktop.
- **Primary action disabled state:** when `onPrimaryAction == null`, the button is disabled (form invalid). Consumer toggles this based on form.validate().
- **Detail dialog action slot:** `actions: List<Widget>?` so consumer can add Edit / Delete / Cancel-Appointment buttons. Library doesn't dictate which actions exist.
</gotchas>

import 'package:flutter/material.dart';
import '../../tokens/spacing.dart';
import 'scheduler_models.dart';
import 'scheduler_time_math.dart';

class EdenSchedulerCreateDialog {
  static Future<T?> show<T>({
    required BuildContext context,
    String title = 'New Appointment',
    required Widget bodySlot,
    String primaryActionLabel = 'Save',
    VoidCallback? onPrimaryAction,
    String cancelLabel = 'Cancel',
  }) {
    final isCompact = MediaQuery.of(context).size.width < 600;
    return showGeneralDialog<T>(
      context: context,
      barrierDismissible: !isCompact,
      barrierLabel: title,
      pageBuilder: (context, _, __) => isCompact
          ? _FullscreenDialog(title: title, body: bodySlot, primaryLabel: primaryActionLabel, onPrimary: onPrimaryAction, cancelLabel: cancelLabel)
          : _ModalDialog(title: title, body: bodySlot, primaryLabel: primaryActionLabel, onPrimary: onPrimaryAction, cancelLabel: cancelLabel),
    );
  }
}

class EdenSchedulerEditDialog {
  static Future<T?> show<T>({
    required BuildContext context,
    required EdenSchedulerEvent initialEvent,
    String? title,
    required Widget bodySlot,
    String primaryActionLabel = 'Save',
    VoidCallback? onPrimaryAction,
    String cancelLabel = 'Cancel',
  }) {
    return EdenSchedulerCreateDialog.show<T>(
      context: context,
      title: title ?? 'Edit Appointment',
      bodySlot: bodySlot,
      primaryActionLabel: primaryActionLabel,
      onPrimaryAction: onPrimaryAction,
      cancelLabel: cancelLabel,
    );
  }
}

class EdenSchedulerDetailDialog {
  static Future<T?> show<T>({
    required BuildContext context,
    required EdenSchedulerEvent event,
    String? title,
    List<Widget> actions = const [],
    Widget? extraBody,
  }) {
    final isCompact = MediaQuery.of(context).size.width < 600;
    final body = Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      mainAxisSize: MainAxisSize.min,
      children: [
        Text(event.title, style: const TextStyle(fontSize: 20, fontWeight: FontWeight.w600)),
        const SizedBox(height: EdenSpacing.space2),
        Text('${EdenSchedulerTimeMath.formatTime(_pad(event.start))} – ${EdenSchedulerTimeMath.formatTime(_pad(event.end))}'),
        if (event.location != null) ...[
          const SizedBox(height: EdenSpacing.space2),
          Text(event.location!.name),
        ],
        if (event.description != null) ...[
          const SizedBox(height: EdenSpacing.space2),
          Text(event.description!),
        ],
        if (extraBody != null) ...[
          const SizedBox(height: EdenSpacing.space2),
          extraBody,
        ],
      ],
    );

    return showGeneralDialog<T>(
      context: context,
      barrierDismissible: !isCompact,
      barrierLabel: title ?? 'Event Details',
      pageBuilder: (context, _, __) => isCompact
          ? _FullscreenDialog(
              title: title ?? 'Event Details',
              body: body,
              primaryLabel: null,
              onPrimary: null,
              cancelLabel: 'Close',
              extraActions: actions,
            )
          : _ModalDialog(
              title: title ?? 'Event Details',
              body: body,
              primaryLabel: null,
              onPrimary: null,
              cancelLabel: 'Close',
              extraActions: actions,
            ),
    );
  }

  static String _pad(DateTime d) =>
      '${d.hour.toString().padLeft(2, '0')}:${d.minute.toString().padLeft(2, '0')}';
}

class _ModalDialog extends StatelessWidget {
  const _ModalDialog({
    required this.title,
    required this.body,
    required this.primaryLabel,
    required this.onPrimary,
    required this.cancelLabel,
    this.extraActions = const [],
  });

  final String title;
  final Widget body;
  final String? primaryLabel;
  final VoidCallback? onPrimary;
  final String cancelLabel;
  final List<Widget> extraActions;

  @override
  Widget build(BuildContext context) {
    return Dialog(
      child: ConstrainedBox(
        constraints: const BoxConstraints(maxWidth: 480),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Padding(
              padding: const EdgeInsets.all(EdenSpacing.space4),
              child: Row(
                children: [
                  Expanded(child: Text(title, style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w600))),
                  IconButton(icon: const Icon(Icons.close), onPressed: () => Navigator.of(context).pop()),
                ],
              ),
            ),
            const Divider(height: 1),
            Flexible(
              child: SingleChildScrollView(
                padding: const EdgeInsets.all(EdenSpacing.space4),
                child: body,
              ),
            ),
            const Divider(height: 1),
            Padding(
              padding: const EdgeInsets.all(EdenSpacing.space2),
              child: Row(
                mainAxisAlignment: MainAxisAlignment.end,
                children: [
                  ...extraActions,
                  TextButton(onPressed: () => Navigator.of(context).pop(), child: Text(cancelLabel)),
                  if (primaryLabel != null) ...[
                    const SizedBox(width: EdenSpacing.space2),
                    FilledButton(onPressed: onPrimary, child: Text(primaryLabel!)),
                  ],
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }
}

class _FullscreenDialog extends StatelessWidget {
  const _FullscreenDialog({
    required this.title,
    required this.body,
    required this.primaryLabel,
    required this.onPrimary,
    required this.cancelLabel,
    this.extraActions = const [],
  });

  final String title;
  final Widget body;
  final String? primaryLabel;
  final VoidCallback? onPrimary;
  final String cancelLabel;
  final List<Widget> extraActions;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        leading: IconButton(icon: const Icon(Icons.close), onPressed: () => Navigator.of(context).pop()),
        title: Text(title),
        actions: [
          ...extraActions,
          if (primaryLabel != null) TextButton(onPressed: onPrimary, child: Text(primaryLabel!)),
        ],
      ),
      body: SingleChildScrollView(padding: const EdgeInsets.all(EdenSpacing.space4), child: body),
    );
  }
}
