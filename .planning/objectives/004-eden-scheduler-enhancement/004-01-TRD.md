```dart
/// A recurrence rule for events. v1 scope: none / daily / weekly. RRULE
/// expansion beyond daily/weekly is deferred — file a follow-up TRD if a
/// downstream app needs monthly / yearly recurrence.
@immutable
class EdenSchedulerEventRecurrence {
  const EdenSchedulerEventRecurrence._({
    required this.kind,
    this.weekdayMask = 0,
    this.until,
  });

  factory EdenSchedulerEventRecurrence.none() =>
      const EdenSchedulerEventRecurrence._(kind: EdenSchedulerRecurrenceKind.none);

  factory EdenSchedulerEventRecurrence.daily({DateTime? until}) =>
      EdenSchedulerEventRecurrence._(
        kind: EdenSchedulerRecurrenceKind.daily,
        until: until,
      );

  factory EdenSchedulerEventRecurrence.weekly({
    required int weekdayMask,
    DateTime? until,
  }) =>
      EdenSchedulerEventRecurrence._(
        kind: EdenSchedulerRecurrenceKind.weekly,
        weekdayMask: weekdayMask,
        until: until,
      );

  final EdenSchedulerRecurrenceKind kind;

  /// Bitmask: Mon=1, Tue=2, Wed=4, Thu=8, Fri=16, Sat=32, Sun=64.
  /// Only meaningful when kind == weekly.
  final int weekdayMask;

  final DateTime? until;

  /// Returns true if [weekday] (1..7, Mon=1) is in the mask. Always false for
  /// non-weekly recurrences.
  bool matchesWeekday(int weekday) {
    if (kind != EdenSchedulerRecurrenceKind.weekly) return false;
    if (weekday < 1 || weekday > 7) return false;
    final bit = 1 << (weekday - 1);
    return (weekdayMask & bit) != 0;
  }
}
```

```dart
import 'package:flutter/foundation.dart';

import 'scheduler_models.dart';

typedef _Clock = DateTime Function();

class EdenSchedulerController extends ChangeNotifier {
  EdenSchedulerController({
    EdenSchedulerView initialView = EdenSchedulerView.week,
    DateTime? initialDate,
    _Clock? clock,
  })  : _clock = clock ?? DateTime.now,
        _view = initialView,
        _focusedDate = _stripTime(initialDate ?? (clock ?? DateTime.now)());

  final _Clock _clock;
  EdenSchedulerView _view;
  DateTime _focusedDate;
  final Set<String> _selectedResources = <String>{};
  final Set<String> _selectedEventIds = <String>{};
  String _search = '';

  EdenSchedulerView get view => _view;
  DateTime get focusedDate => _focusedDate;
  DateTime get today => _stripTime(_clock());
  Set<String> get selectedResources => Set.unmodifiable(_selectedResources);
  Set<String> get selectedEventIds => Set.unmodifiable(_selectedEventIds);
  String get search => _search;

  static DateTime _stripTime(DateTime d) => DateTime(d.year, d.month, d.day);

  void setView(EdenSchedulerView v) {
    if (v == _view) return;
    _view = v;
    notifyListeners();
  }

  void setFocusedDate(DateTime d) {
    final stripped = _stripTime(d);
    if (stripped == _focusedDate) return;
    _focusedDate = stripped;
    notifyListeners();
  }

  void navigate(int delta) {
    switch (_view) {
      case EdenSchedulerView.month:
        _focusedDate = DateTime(_focusedDate.year, _focusedDate.month + delta, 1);
        break;
      case EdenSchedulerView.week:
        _focusedDate = _focusedDate.add(Duration(days: 7 * delta));
        break;
      case EdenSchedulerView.workWeek:
        _focusedDate = _focusedDate.add(Duration(days: 7 * delta));
        break;
      case EdenSchedulerView.day:
        _focusedDate = _focusedDate.add(Duration(days: delta));
        break;
      case EdenSchedulerView.list:
        _focusedDate = _focusedDate.add(Duration(days: 7 * delta));
        break;
      case EdenSchedulerView.mobile:
        _focusedDate = _focusedDate.add(Duration(days: delta));
        break;
      case EdenSchedulerView.swimlane:
        _focusedDate = _focusedDate.add(Duration(days: 7 * delta));
        break;
    }
    notifyListeners();
  }

  void goToday() {
    final t = today;
    if (t == _focusedDate) return;
    _focusedDate = t;
    notifyListeners();
  }

  void toggleResourceFilter(String resourceId) {
    if (_selectedResources.contains(resourceId)) {
      _selectedResources.remove(resourceId);
    } else {
      _selectedResources.add(resourceId);
    }
    notifyListeners();
  }

  void setSearch(String value) {
    final trimmed = value.trim();
    if (trimmed == _search) return;
    _search = trimmed;
    notifyListeners();
  }

  void toggleSelection(String eventId) {
    if (_selectedEventIds.contains(eventId)) {
      _selectedEventIds.remove(eventId);
    } else {
      _selectedEventIds.add(eventId);
    }
    notifyListeners();
  }

  void clearSelection() {
    if (_selectedEventIds.isEmpty) return;
    _selectedEventIds.clear();
    notifyListeners();
  }
}
```
