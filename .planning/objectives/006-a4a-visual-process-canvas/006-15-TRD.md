      data: const {'nodeType': 'orphan', 'orphanType': 'task', 'displayName': 'Inspect machinery', 'config': {'runtimeComponent': 'checklist'}},
    ));

    _controller.dispose();
    super.dispose();

  void _resetDemo() {
    setState(() {
      _controller.reset();
      _definition = _initialDemo();
      _controller.addOrphan(/* re-seed orphan */);
    });
  }

  void _setLayout(EdenProcessLayoutEngine engine) {
    setState(() {
      _layout = engine;
      _controller.layout = engine;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Process Builder Demo'),
        actions: [
          DropdownButton<String>(
            value: _layout.id,
            items: const [
              DropdownMenuItem(value: 'swimlane', child: Text('Swimlane')),
              DropdownMenuItem(value: 'free_form', child: Text('Free-form')),
              DropdownMenuItem(value: 'grid', child: Text('Grid')),
              DropdownMenuItem(value: 'linear', child: Text('Linear')),

              switch (v) {
                case 'swimlane': _setLayout(const EdenSwimlaneLayout()); break;

    );
