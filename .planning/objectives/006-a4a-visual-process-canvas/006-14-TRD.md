1. Build diagram nodes + edges via `EdenProcessGraphBuilder.build(definition, controller: controller)`.
2. Apply layout via `controller.layout.applyLayout(nodes, edges)`.

4. Merge with `controller.userEdges`.
5. Pass to `EdenDiagram(data: ..., customNodeRenderer: EdenProcessNodeRenderer.dispatch, dropTargetNodeId: _dragOverNodeId, onDropTargetChanged: ...)`.

**Drop dispatch (donor `handleDrop` at VisualProcessCanvas.tsx:516-579):**
```dart
void _onDrop(EdenProcessDragPayload payload, Offset screenPos) {
  // Use the canvas state's hitTestNode (TRD 02) to find the drop target
  final target = _diagramKey.currentState?.hitTestNode(screenPos);
  final config = payload.config;
  switch (payload.dragType) {
    case 'phase':
      widget.onAddPhase?.call();
      break;
    case 'task':
      if (target?.data['nodeType'] == 'taskGroup') {
        widget.onAddTask?.call(target!.data['groupId'] as int, config);
      } else if (target?.data['nodeType'] == 'phase') {
        // Add to FIRST group of phase, OR create default group if none
        final phaseId = target!.data['phaseId'] as int;
        // Donor adds to first group via onAddTask(phaseId, config) — let consumer decide
        widget.onAddTaskInPhase?.call(phaseId, config);
      } else {
        // empty canvas → first phase
        final firstPhaseId = widget.definition.phases.firstOrNull?.id;
        if (firstPhaseId != null) widget.onAddTaskInPhase?.call(firstPhaseId, config);
      }
      break;
    case 'taskGroup':
    case 'templateGroup':
      final phaseId = target?.data['nodeType'] == 'phase'
        ? target!.data['phaseId'] as int
        : widget.definition.phases.firstOrNull?.id;
      if (phaseId != null) {
        if (config?['templateId'] != null) {
          widget.onImportTemplate?.call(phaseId, config!['templateId'] as int);
        } else {
          widget.onAddTaskGroup?.call(phaseId);
        }
      }
      break;
  }
}
```

```dart
EdenDiagramEdge? hitTestEdge(Offset screenPosition) {
  final canvasPos = _toCanvas(screenPosition);
  for (int i = widget.data.edges.length - 1; i >= 0; i--) {
    final edge = widget.data.edges[i];
    final source = widget.data.nodeById(edge.sourceId);
    final target = widget.data.nodeById(edge.targetId);
    if (source == null || target == null) continue;
    final start = source.portOffset(edge.sourcePort);
    final end = target.portOffset(edge.targetPort);
    if (_distanceToSegment(canvasPos, start, end) < 8) return edge;
  }
  return null;
}

double _distanceToSegment(Offset p, Offset a, Offset b) {
  final dx = b.dx - a.dx;
  final dy = b.dy - a.dy;
  if (dx == 0 && dy == 0) return (p - a).distance;
  final t = ((p.dx - a.dx) * dx + (p.dy - a.dy) * dy) / (dx * dx + dy * dy);
  final tClamped = t.clamp(0.0, 1.0);
  final projection = Offset(a.dx + tClamped * dx, a.dy + tClamped * dy);
  return (p - projection).distance;
}
```
