- The focus ring + default border conflict: pick ONE based on focus state. Stacking them creates a "thick border" visual that looks like a bug.

   **Cleaner pattern:** make `_InteractiveEdenCard` a separate `StatefulWidget` that does NOT extend EdenCard but RETURNS an EdenCard internally OR fully composes the visual itself. Since `EdenCard.interactive(...)` is a factory, the return type must be assignable to EdenCard. Use this pattern:
   ```dart
   class EdenCard extends StatelessWidget {
     // ... existing fields unchanged

     const EdenCard({ super.key, ... existing params, this.onLongPress, this.hoverLift = false, this.focusRing = false, })
       : assert(...);

     factory EdenCard.interactive({...}) {
       return EdenCard(
         key: key,
         child: child, title: title, subtitle: subtitle, image: image, horizontal: horizontal, padding: padding,
         onTap: onTap, onLongPress: onLongPress, borderColor: borderColor,
         hoverLift: true, focusRing: true,
       );
     }

     final bool hoverLift;
     final bool focusRing;
     final VoidCallback? onLongPress;
     // ... existing fields

     @override
     Widget build(BuildContext context) {
       if (hoverLift || focusRing) {
         return _InteractiveBody(this);  // Stateful wrapper that handles animation + focus
       }
       return _buildContent(context);  // existing default path
     }
   }

   class _InteractiveBody extends StatefulWidget {
     final EdenCard card;
     const _InteractiveBody(this.card);
     @override
     State<_InteractiveBody> createState() => _InteractiveBodyState();
   }

   class _InteractiveBodyState extends State<_InteractiveBody> with TickerProviderStateMixin {
     late final AnimationController _liftController;
     bool _focused = false;

     @override
     void initState() { super.initState(); _liftController = AnimationController.unbounded(vsync: this); }
     @override
     void dispose() { _liftController.dispose(); super.dispose(); }

     void _setHover(bool hover) {
       _liftController.animateWith(EdenSprings.simulationFor(
         EdenSprings.smooth, from: _liftController.value, to: hover ? 1.0 : 0.0,
       ));
     }

     @override
     Widget build(BuildContext context) {
       return FocusableActionDetector(
         onShowFocusHighlight: (f) => setState(() => _focused = f),
         onShowHoverHighlight: _setHover,
         child: AnimatedBuilder(
           animation: _liftController,
           builder: (_, __) {
             return Transform.translate(
               offset: Offset(0, -2 * _liftController.value.clamp(0.0, 1.0)),
               child: ConstrainedBox(
                 constraints: const BoxConstraints(minHeight: 44),
                 child: Semantics(button: true, label: widget.card.title ?? 'Card', focused: _focused,
                   child: widget.card._buildContent(context, lifted: _liftController.value, focused: _focused),
                 ),
               ),
             );
           },
         ),
       );
     }
   }
   ```
