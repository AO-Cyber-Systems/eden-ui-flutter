import 'package:flutter/foundation.dart' show defaultTargetPlatform, kIsWeb;
import 'package:flutter/material.dart';

import '../../eden_diagram/eden_diagram_exports.dart';
import '../workflow_action_registry.dart';
import '../workflow_action_field_spec.dart';

class EdenActionNode extends StatefulWidget {
  const EdenActionNode({super.key, required this.context});
  final EdenDiagramNodeContext context;
  @override State<EdenActionNode> createState() => _EdenActionNodeState();
}

class _EdenActionNodeState extends State<EdenActionNode> {
  bool _hovered = false;

  bool get _isTouch => !kIsWeb &&
    (defaultTargetPlatform == TargetPlatform.iOS ||
     defaultTargetPlatform == TargetPlatform.android);

  String get _actionTypeId => widget.context.node.data['actionType'] as String? ?? 'create_task';
  Map<String, dynamic> get _config =>
    (widget.context.node.data['config'] as Map?)?.cast<String, dynamic>() ?? const {};

  EdenWorkflowActionType? get _actionType =>
    EdenWorkflowActionRegistry.instance.lookup(_actionTypeId);

  String _configSummary() => switch (_actionTypeId) {
    'create_task' => _config['title'] as String? ?? '',
    'send_notification' => _config['title'] as String? ?? '',
    'create_callback' => _config['reason'] as String? ?? '',
    'update_status' => _config['newStatus'] != null ? '→ ${_config['newStatus']}' : '',
    'send_email' => _config['to'] as String? ?? '',
    'send_sms' => _config['phoneNumber'] as String? ?? '',
    _ => '',
  };

  Future<void> _openEditor() async {
    var newType = _actionTypeId;
    var newConfig = Map<String, dynamic>.from(_config);

    final saved = await showDialog<bool>(
      context: context,
      builder: (dialogCtx) {
        return StatefulBuilder(builder: (stateCtx, setLocal) {
          final actionType = EdenWorkflowActionRegistry.instance.lookup(newType);
          return AlertDialog(
            content: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 360, maxHeight: 400),
              child: SingleChildScrollView(child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  const Text('Action Type', style: TextStyle(fontSize: 12, fontWeight: FontWeight.w500)),
                  DropdownButtonFormField<String>(
                    value: newType,
                    items: EdenWorkflowActionRegistry.instance.all().map((t) =>
                      DropdownMenuItem(value: t.id, child: Text(t.displayName))).toList(),
                    onChanged: (v) {
                      if (v == null) return;
                      setLocal(() {
                        newType = v;
                        newConfig = {};  // reset config on action-type change
                      });
                    },
                  ),
                  const SizedBox(height: 12),
                  // Dynamic field rendering
                  for (final field in actionType?.fields ?? const <EdenWorkflowActionFieldSpec>[])
                    Padding(
                      padding: const EdgeInsets.only(bottom: 8),
                      child: _buildField(field, newConfig, (key, value) {
                        setLocal(() => newConfig[key] = value);
                      }),
                    ),
                ],
              )),
            ),
            actions: [
              TextButton(onPressed: () => Navigator.of(dialogCtx).pop(false), child: const Text('Cancel')),
              FilledButton(onPressed: () => Navigator.of(dialogCtx).pop(true), child: const Text('Save')),
            ],
          );
        });
      },
    );
    if (saved == true) {
      widget.context.onUpdate?.call({
        'actionType': newType,
        'config': newConfig,
        'label': EdenWorkflowActionRegistry.instance.lookup(newType)?.displayName ?? newType,
      });
    }
  }

  Widget _buildField(EdenWorkflowActionFieldSpec spec, Map<String, dynamic> config, void Function(String, dynamic) setValue) {
    final current = config[spec.id];
    switch (spec.type) {
      case 'textarea':
        return TextFormField(
          initialValue: current as String? ?? '',
          maxLines: 3,
          decoration: InputDecoration(labelText: spec.label, isDense: true, border: const OutlineInputBorder()),
          onChanged: (v) => setValue(spec.id, v),
        );
      case 'select':
        return DropdownButtonFormField<String>(
          value: current as String? ?? (spec.options.isNotEmpty ? spec.options.first.value : null),
          decoration: InputDecoration(labelText: spec.label, isDense: true, border: const OutlineInputBorder()),
          items: spec.options.map((o) => DropdownMenuItem(value: o.value, child: Text(o.label))).toList(),
          onChanged: (v) => setValue(spec.id, v),
        );
      case 'number':
        return TextFormField(
          initialValue: (current ?? '').toString(),
          keyboardType: TextInputType.number,
          decoration: InputDecoration(labelText: spec.label, isDense: true, border: const OutlineInputBorder()),
          onChanged: (v) => setValue(spec.id, int.tryParse(v)),
        );
      default:  // 'text'
        return TextFormField(
          initialValue: current as String? ?? '',
          decoration: InputDecoration(labelText: spec.label, isDense: true, border: const OutlineInputBorder()),
          onChanged: (v) => setValue(spec.id, v),
        );
    }
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
