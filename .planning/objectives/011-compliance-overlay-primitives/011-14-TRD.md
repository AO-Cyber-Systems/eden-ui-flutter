- **Native vs English label.** Some users only recognize their language in native form (中文 vs Chinese). USWDS spec says use native label primarily, English label secondary in tooltip.
- **Selection highlighting.** In dropdown: use `selected: true` styling. In chip row: chip backgroundColor bolder when selected.
- **Hand-built fixtures** — header required.
- **Test-pairing rule** — paired test required.
- **APPEND only for uswds_screen.dart.**
</gotchas>

import 'package:flutter/material.dart';

enum EdenLanguageSelectorLayout { button, chips }

@immutable
class EdenLanguageOption {
  const EdenLanguageOption({
    required this.code,
    required this.nativeLabel,
    required this.englishLabel,
  });
  final String code;
  final String nativeLabel;
  final String englishLabel;

  @override
  bool operator ==(Object other) =>
      other is EdenLanguageOption &&
      other.code == code &&
      other.nativeLabel == nativeLabel &&
      other.englishLabel == englishLabel;

  @override
  int get hashCode => Object.hash(code, nativeLabel, englishLabel);
}

class EdenLanguageOptions {
  EdenLanguageOptions._();

  /// USWDS / EO 13166 default — 7 most-spoken non-English languages in U.S.
  /// federal contexts per Census Bureau Language Use Tables.
  static const usFederalDefault = <EdenLanguageOption>[
    EdenLanguageOption(code: 'en', nativeLabel: 'English', englishLabel: 'English'),
    EdenLanguageOption(code: 'es', nativeLabel: 'Español', englishLabel: 'Spanish'),
    EdenLanguageOption(code: 'zh', nativeLabel: '中文', englishLabel: 'Chinese'),
    EdenLanguageOption(code: 'vi', nativeLabel: 'Tiếng Việt', englishLabel: 'Vietnamese'),
    EdenLanguageOption(code: 'ko', nativeLabel: '한국어', englishLabel: 'Korean'),
    EdenLanguageOption(code: 'ru', nativeLabel: 'Русский', englishLabel: 'Russian'),
    EdenLanguageOption(code: 'ar', nativeLabel: 'العربية', englishLabel: 'Arabic'),
  ];
}

/// USWDS language toggle pattern. Required for federal i18n mandate (EO
/// 13166) + Section 508. Two layouts: [button] (compact) and [chips]
/// (horizontal row).
///
/// Widget is selection-only — consumer wires app locale change in
/// [onLanguageChanged] callback.
class EdenLanguageSelector extends StatelessWidget {
  const EdenLanguageSelector({
    super.key,
    required this.current,
    required this.onLanguageChanged,
    this.options = EdenLanguageOptions.usFederalDefault,
    this.layout = EdenLanguageSelectorLayout.button,
  });

  final EdenLanguageOption current;
  final ValueChanged<EdenLanguageOption> onLanguageChanged;
  final List<EdenLanguageOption> options;
  final EdenLanguageSelectorLayout layout;

  @override
  Widget build(BuildContext context) {
    return Semantics(
      label: 'Language selector, current: ${current.nativeLabel}',
      container: true,
      child: layout == EdenLanguageSelectorLayout.button
          ? _buttonLayout(context)
          : _chipsLayout(context),
    );
  }

  Widget _buttonLayout(BuildContext context) {
    return PopupMenuButton<EdenLanguageOption>(
      onSelected: onLanguageChanged,
      tooltip: 'Select language',
      itemBuilder: (context) => [
        for (final opt in options)
          PopupMenuItem(
            value: opt,
            child: Semantics(
              label: 'Switch to ${opt.nativeLabel}',
              child: Row(
                children: [
                  if (opt.code == current.code) const Icon(Icons.check, size: 16),
                  if (opt.code == current.code) const SizedBox(width: 8),
                  Text(opt.nativeLabel),
                  if (opt.englishLabel != opt.nativeLabel) ...[
                    const SizedBox(width: 8),
                    Text('(${opt.englishLabel})', style: const TextStyle(color: Colors.grey, fontSize: 12)),
                  ],
                ],
              ),
            ),
          ),
      ],
      child: Padding(
        padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 8),
        child: Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            const Icon(Icons.language, size: 16),
            const SizedBox(width: 6),
            Text(current.nativeLabel),
            const SizedBox(width: 4),
            const Icon(Icons.arrow_drop_down, size: 16),
          ],
        ),
      ),
    );
  }

  Widget _chipsLayout(BuildContext context) {
    return Wrap(
      spacing: 8,
      runSpacing: 8,
      children: [
        for (final opt in options)
          Semantics(
            label: 'Switch to ${opt.nativeLabel}',
            button: true,
            child: ChoiceChip(
              label: Text(opt.nativeLabel),
              selected: opt.code == current.code,
              onSelected: (sel) {
                if (sel) onLanguageChanged(opt);
              },
            ),
          ),
      ],
    );
  }
}
