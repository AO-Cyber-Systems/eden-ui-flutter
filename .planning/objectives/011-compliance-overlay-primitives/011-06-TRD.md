- **Do NOT bind to DHHS / DOJ / HIPAA-specific APIs.** Generic value classes only.

</anti_patterns>

<gotchas>
- **6 tabs at 390pt.** TabBar with `isScrollable: true` so tabs scroll horizontally on narrow.
- **TabController lifecycle.** Use `DefaultTabController(length: 6, child: ...)` OR manage your own. DefaultTabController is simpler for a stateless shell.
- **Empty data per tab.** If `documents == null || documents.isEmpty` → render "No documents yet" empty state. Same for contacts, notes, activity.
- **Audit tab pass-through.** When `auditEntries == null`, render "No audit log available". When provided, just wrap EdenAuditLogViewer.
- **Privileged notes.** `isPrivileged: true` renders with a small `EdenClassificationBanner` inline (custom level, labelText: 'PRIVILEGED') — civilian re-use of 011-01.
- **Hand-built fixtures.** Required header.
- **Test-pairing rule** — paired test required.
- **APPEND only for compliance_screen.dart.**
</gotchas>

/// Multi-tab regulated dossier shell for DHHS social services, DOJ case
/// files, IG investigations, medical patient charts, legal matters,
/// insurance claims, and complex commercial CRM dossiers.
///
/// Composes [EdenClassificationBannerScaffold] for classification overlay
/// and [EdenAuditLogViewer] for the Audit tab.
class EdenCaseFileShell extends StatelessWidget {
  const EdenCaseFileShell({super.key, required this.data});

  final EdenCaseFileData data;

  @override
  Widget build(BuildContext context) {
    final body = DefaultTabController(
      length: 6,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          _Header(data: data),
          Container(
            color: Theme.of(context).colorScheme.surfaceContainerHighest,
            child: const TabBar(
              isScrollable: true,
              tabs: [
                Tab(text: 'Overview'),
                Tab(text: 'Activity'),
                Tab(text: 'Documents'),
                Tab(text: 'Contacts'),
                Tab(text: 'Notes'),
                Tab(text: 'Audit'),
              ],
            ),
          ),
          Expanded(
            child: TabBarView(
              children: [
                _OverviewTab(data: data),
                _ActivityTab(entries: data.activityEntries),
                _DocumentsTab(docs: data.documents),
                _ContactsTab(contacts: data.contacts),
                _NotesTab(notes: data.notes),
                _AuditTab(entries: data.auditEntries),
              ],
            ),
          ),
        ],
      ),
    );

    if (data.classification != null) {
      return EdenClassificationBannerScaffold(
        level: data.classification!,
        child: body,
      );
    }
    return body;
  }
}

class _Header extends StatelessWidget {
  const _Header({required this.data});
  final EdenCaseFileData data;

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              Expanded(
                child: Text(data.caseTitle, style: Theme.of(context).textTheme.titleLarge),
              ),
              Container(
                padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 4),
                decoration: BoxDecoration(color: Colors.blue.shade100, borderRadius: BorderRadius.circular(12)),
                child: Text(data.status, style: TextStyle(color: Colors.blue.shade900, fontSize: 11, fontWeight: FontWeight.w700)),
              ),
            ],
          ),
          const SizedBox(height: 4),
          Text(data.caseId, style: const TextStyle(fontFamily: 'monospace', color: Colors.grey)),
        ],
      ),
    );
  }
}

class _OverviewTab extends StatelessWidget {
  const _OverviewTab({required this.data});
  final EdenCaseFileData data;

  @override
  Widget build(BuildContext context) {
    return SingleChildScrollView(
      padding: const EdgeInsets.all(16),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          _kv('Opened', data.openedAt.toIso8601String().split('T').first),
          if (data.assignedTo != null) _kv('Assigned to', data.assignedTo!),
          const Divider(),
          for (final entry in data.headerMetadata.entries) _kv(entry.key, entry.value),
        ],
      ),
    );
  }

  Widget _kv(String k, String v) => Padding(
    padding: const EdgeInsets.symmetric(vertical: 4),
    child: Row(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        SizedBox(width: 120, child: Text(k, style: const TextStyle(fontWeight: FontWeight.w600, fontSize: 13))),
        Flexible(child: Text(v, style: const TextStyle(fontSize: 13))),
      ],
    ),
  );
}

class _ActivityTab extends StatelessWidget {
  const _ActivityTab({required this.entries});
  final List<EdenActivity>? entries;

  @override
  Widget build(BuildContext context) {
    final list = entries;
    if (list == null || list.isEmpty) {
      return const Center(child: Padding(padding: EdgeInsets.all(16), child: Text('No activity yet')));
    }
    return ListView.builder(
      itemCount: list.length,
      itemBuilder: (context, i) {
        final a = list[i];
        return ListTile(
          leading: const Icon(Icons.history, size: 18),
          title: Text(a.title, style: const TextStyle(fontSize: 13)),
          subtitle: Text('${a.actorName} · ${a.timestamp.toIso8601String().split('T').first}', style: const TextStyle(fontSize: 11)),
        );
      },
    );
  }
}

class _DocumentsTab extends StatelessWidget {
  const _DocumentsTab({required this.docs});
  final List<EdenCaseFileDocument>? docs;

  @override
  Widget build(BuildContext context) {

    if (list == null || list.isEmpty) {
      return const Center(child: Padding(padding: EdgeInsets.all(16), child: Text('No documents yet')));

    return ListView.builder(
      itemCount: list.length,
      itemBuilder: (context, i) {
        final d = list[i];
        return ListTile(
          leading: const Icon(Icons.description_outlined),
          title: Text(d.name, style: const TextStyle(fontSize: 13)),
          subtitle: Text('${(d.sizeBytes / 1024).toStringAsFixed(1)} KB · by ${d.uploadedBy} · ${d.uploadedAt.toIso8601String().split('T').first}', style: const TextStyle(fontSize: 11)),
          trailing: Text(d.status.name, style: const TextStyle(fontSize: 11, color: Colors.grey)),
        );
      },
    );
  }
}

              ListTile(

                title: Text('${entry.key}: ${entry.value}', style: const TextStyle(fontSize: 12)),

        );
      },
    );
  }
}
