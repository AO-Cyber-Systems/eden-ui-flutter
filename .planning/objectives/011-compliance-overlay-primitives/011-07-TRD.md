Justification dialog:
```dart
class _BreakGlassJustificationDialog extends StatefulWidget {
  const _BreakGlassJustificationDialog({required this.roleId, required this.permissionId});
  final String roleId;
  final String permissionId;
  @override
  State<_BreakGlassJustificationDialog> createState() => _BreakGlassJustificationDialogState();
}

class _BreakGlassJustificationDialogState extends State<_BreakGlassJustificationDialog> {
  String _justification = '';
  bool get _canConfirm => _justification.trim().length >= 20;

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      title: const Text('Break-glass override'),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text('You are about to override permissions for ${widget.permissionId} on ${widget.roleId}.'),
          const SizedBox(height: 12),
          TextField(
            decoration: const InputDecoration(
              labelText: 'Justification',
              helperText: 'Minimum 20 characters. This action will be logged.',
              border: OutlineInputBorder(),
            ),
            maxLines: 3,
            onChanged: (v) => setState(() => _justification = v),
          ),
        ],
      ),
      actions: [
        TextButton(onPressed: () => Navigator.of(context).pop(), child: const Text('Cancel')),
        ElevatedButton(
          onPressed: _canConfirm ? () => Navigator.of(context).pop(_justification) : null,
          child: const Text('Confirm'),
        ),
      ],
    );
  }
}
```

Helper to launch + capture result:
```dart
Future<void> _triggerBreakGlass(BuildContext context, String roleId, String permissionId) async {
  final result = await showDialog<String>(
    context: context,
    builder: (_) => _BreakGlassJustificationDialog(roleId: roleId, permissionId: permissionId),
  );
  if (result != null && widget.onBreakGlass != null) {
    widget.onBreakGlass!(roleId, permissionId, result);
  }
}
```
